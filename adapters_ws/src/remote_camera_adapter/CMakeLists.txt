cmake_minimum_required(VERSION 3.16)
project(remote_camera_adapter)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(robot_task_interfaces REQUIRED)
find_package(perception_task_adapter REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs aruco)
find_package(Threads REQUIRED)
add_executable(remote_camera_adapter src/main.cpp)
target_compile_features(remote_camera_adapter PRIVATE cxx_std_17)
target_compile_options(remote_camera_adapter PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(remote_camera_adapter SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(remote_camera_adapter rclcpp rclcpp_action robot_task_interfaces)
target_link_libraries(remote_camera_adapter perception_task_adapter::perception_task_adapter_node_lib ${OpenCV_LIBS} Threads::Threads)
install(TARGETS remote_camera_adapter RUNTIME DESTINATION lib/${PROJECT_NAME})
ament_package()
